Here is the complete implementation tailored to your exact `public.leads` schema.

When a user submits their email, the Supabase Edge Function saves them to `public.leads`, adds them to Brevo, and sends them a **"Welcome to our Newsletter at South African Funeral Supplies"** email with catalog links.

---

### Step 1: The Supabase Edge Function

Create (or update) this file in your codebase:
📁 **`supabase/functions/newsletter-subscribe/index.ts`**

This function matches your database schema columns (`id`, `email`, `first_name`, `source`, `utm_source`, `utm_medium`, `utm_campaign`, `lead_magnet`, `brevo_contact_id`, `email_sent`, `created_at`). It embeds the welcome email directly so it sends via Brevo.

serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const payload: SubscribeRequest = await req.json();

    // 1. Honeypot check for bots
    if (payload.honeypot && payload.honeypot.trim() !== "") {
      return new Response(JSON.stringify({ success: true }), {
        headers: { ...corsHeaders, "Content-Type": "application/json" },
        status: 200,
      });
    }

    // 2. Validate email
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!payload.email || !emailRegex.test(payload.email.trim())) {
      return new Response(
        JSON.stringify({ success: false, error: "Please provide a valid email address." }),
        { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 400 }
      );
    }

    const cleanEmail = payload.email.trim().toLowerCase();
    const firstName = payload.first_name?.trim() || "";

    // 3. Supabase Client
    const supabaseUrl = Deno.env.get("SUPABASE_URL")!;
    const supabaseServiceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
    const supabase = createClient(supabaseUrl, supabaseServiceKey);

    // 4. Check if lead already exists
    const { data: existingLead } = await supabase
      .from("leads")
      .select("id, email, brevo_contact_id")
      .eq("email", cleanEmail)
      .maybeSingle();

    let leadId = existingLead?.id;
    let isDuplicate = !!existingLead;

    if (!existingLead) {
      // Insert new lead matching your exact public.leads table
      const { data: newLead, error: insertError } = await supabase
        .from("leads")
        .insert({
          email: cleanEmail,
          first_name: firstName || null,
          source: payload.source || "catalog-page",
          utm_source: payload.utm_source || null,
          utm_medium: payload.utm_medium || null,
          utm_campaign: payload.utm_campaign || null,
          lead_magnet: payload.lead_magnet || "catalog-welcome",
          email_sent: false,
        })
        .select("id")
        .single();

      if (insertError) {
        throw new Error(`Database error: ${insertError.message}`);
      }
      leadId = newLead.id;
    }

    // 5. Brevo Integration: Add Contact & Send Welcome Email
    const brevoApiKey = Deno.env.get("BREVO_API_KEY");
    const brevoListId = Number(Deno.env.get("BREVO_LIST_ID") || 3);
    let brevoContactId: number | null = existingLead?.brevo_contact_id || null;
    let emailSentSuccess = false;

    if (brevoApiKey) {
      // Step A: Sync to Brevo Contacts List
      try {
        const contactRes = await fetch("https://api.brevo.com/v3/contacts", {
          method: "POST",
          headers: {
            "api-key": brevoApiKey,
            "Content-Type": "application/json",
            "Accept": "application/json",
          },
          body: JSON.stringify({
            email: cleanEmail,
            attributes: {
              FIRSTNAME: firstName,
              SOURCE: payload.source || "catalog-page",
            },
            listIds: [brevoListId],
            updateEnabled: true,
          }),
        });

        const contactData = await contactRes.json();
        if (contactData?.id) {
          brevoContactId = contactData.id;
        }
      } catch (err) {
        console.error("Brevo contact sync error:", err);
      }

      // Step B: Send Transactional "Welcome to our Newsletter" Email
      try {
        const welcomeEmailHtml = generateWelcomeEmailHtml(firstName);
        const emailRes = await fetch("https://api.brevo.com/v3/smtp/email", {
          method: "POST",
          headers: {
            "api-key": brevoApiKey,
            "Content-Type": "application/json",
            "Accept": "application/json",
          },
          body: JSON.stringify({
            sender: {
              name: "South African Funeral Supplies",
              email: "sales@safuneral.co.za", // Must be verified in your Brevo senders
            },
            to: [{ email: cleanEmail, name: firstName || "Trade Partner" }],
            subject: "Welcome to South African Funeral Supplies | Official Trade Catalog",
            htmlContent: welcomeEmailHtml,
          }),
        });

        emailSentSuccess = emailRes.ok;
      } catch (err) {
        console.error("Brevo email sending error:", err);
      }

      // Step C: Update public.leads record
      if (leadId) {
        await supabase
          .from("leads")
          .update({
            brevo_contact_id: brevoContactId,
            email_sent: emailSentSuccess,
          })
          .eq("id", leadId);
      }
    }

    return new Response(
      JSON.stringify({
        success: true,
        status: isDuplicate ? "already_subscribed" : "subscribed",
        message: "Welcome to our newsletter at South African Funeral Supplies! Check your inbox for your welcome bulletin.",
      }),
      {
        headers: { ...corsHeaders, "Content-Type": "application/json" },
        status: 200,
      }
    );
  } catch (error: any) {
    return new Response(
      JSON.stringify({ success: false, error: error.message || "An unexpected error occurred." }),
      {
        headers: { ...corsHeaders, "Content-Type": "application/json" },
        status: 500,
      }
    );
  }
});

---

### Step 2: Deploy to Supabase

1. Set your **Brevo API key** in Supabase:
```bash
npx supabase secrets set BREVO_API_KEY="xkeysib-your-real-brevo-api-key"
npx supabase secrets set BREVO_LIST_ID="3"
```

2. Deploy the function:
```bash
npx supabase functions deploy newsletter-subscribe --no-verify-jwt
```
